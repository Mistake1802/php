function doimk(){
    let u = user.value;
    let o = old.value;
    let n = new.value;

    if(u==""||o==""||n==""){
        msg.innerHTML="Nhập đủ!";
        return;
    }

    let xhr = new XMLHttpRequest();
    xhr.open("POST","xuly.php",true);
    xhr.setRequestHeader("Content-type","application/x-www-form-urlencoded");

    xhr.onload = function(){
        msg.innerHTML = this.responseText;
    };

    xhr.send("u="+u+"&o="+o+"&n="+n);
}
</script>

<?php
$conn = new mysqli("localhost","root","","accountDB");

$u = $_POST['u'];

$n = $_POST['n'];

// kiểm tra mật khẩu cũ
$sql = "SELECT * FROM users WHERE username='$u' AND password='$o'";
$result = $conn->query($sql);

if($result->num_rows > 0){
    // đúng → update
    $conn->query("UPDATE users SET password='$n' WHERE username='$u'");
    echo "Đổi mật khẩu thành công!";
}else{
    echo "Sai mật khẩu cũ!";
}

$conn->close();
?>

<script>
function load(){
    let xhr = new XMLHttpRequest();
    xhr.open("GET","xuly.php",true);

    xhr.onload = function(){
        list.innerHTML = this.responseText;
    };

    xhr.send();
}

function gui(){
    let nd = document.getElementById("nd").value;

    if(nd=="") return;

    let xhr = new XMLHttpRequest();
    xhr.open("POST","xuly.php",true);
    xhr.setRequestHeader("Content-type","application/x-www-form-urlencoded");

    xhr.onload = function(){
        document.getElementById("nd").value="";
        load(); // reload list
    };

    xhr.send("nd="+nd);
}

// load lần đầu
load();
</script>

</body>
</html>

<?php
$conn = new mysqli("localhost","root","","commentDB");

// nếu POST → thêm comment

    $nd = $_POST['nd'];
    $conn->query("INSERT INTO comment(noidung) VALUES('$nd')");

// luôn hiển thị list
$result = $conn->query("SELECT * FROM comment ORDER BY id DESC");
