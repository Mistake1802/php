<input type="password" id="repass" placeholder="Nhập lại">
<input type="text" id="email" placeholder="Email">

<button onclick="dangky()">Đăng ký</button>

<div id="msg"></div>
</div>

<script>
function dangky(){
    let u = user.value;
    let p = pass.value;
    let r = repass.value;
    let e = email.value;

    if(u==""||p==""||r==""||e==""){
        msg.innerHTML="Nhập đủ!";
        return;
    }

    if(p.length < 6){
        msg.innerHTML="Pass >=6 ký tự";
        return;
    }

    if(p != r){
        msg.innerHTML="Pass không khớp";
        return;
    }

    let xhr = new XMLHttpRequest();
    xhr.open("POST","xuly.php",true);
    xhr.setRequestHeader("Content-type","application/x-www-form-urlencoded");

    xhr.onload = function(){
        msg.innerHTML = this.responseText;
    };

    xhr.send("u="+u+"&p="+p+"&e="+e);
}
</script>

</body>
</html>
